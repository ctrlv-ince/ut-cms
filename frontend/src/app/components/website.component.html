// Retrieve moving average historical buffer from flow context
let history = flow.get('temp_history') || [];
history.push(msg.payload.ppm);
if (history.length > 5) history.shift();
flow.set('temp_history', history);

// Compute smoothed average
let sum = history.reduce((a, b) => a + b, 0);
msg.smoothed_ppm = sum / history.length;
msg.is_hazard = (msg.smoothed_ppm > 500);

return msg;
